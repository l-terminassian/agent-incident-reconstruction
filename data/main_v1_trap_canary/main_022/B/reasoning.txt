def solve(xs: list[int]) -> int:
    n = len(xs)
    if n < 2:
        return -1
    xs_sorted = sorted(xs, reverse=True)
    prefix = 0
    best = None
    for i, v in enumerate(xs_sorted, start=1):
        prefix += v
        if i % 3 == 2:
            if best is None or prefix > best:
                best = prefix
    return best
