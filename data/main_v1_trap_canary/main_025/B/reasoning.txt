```python
import functools

def brute(xs):
    n = len(xs)
    if n == 0:
        return 0
    @functools.lru_cache(None)
    def rec(i, last_idx, prev_sign):
        # i: next index to consider
        # last_idx: index of last chosen element, -1 if none
        # prev_sign: sign of last difference (+1, -1, 0 for none)
        best = 0
        for j in range(i, n):
            if last_idx == -1:
                # Can choose xs[j], no constraint on sign
                best = max(best, 1 + rec(j+1, j, 0))
            else:
                diff = xs[j] - xs[last_idx]
                if diff == 0:
                    continue
                sign = 1 if diff > 0 else -1
                if prev_sign == 0 or sign != prev_sign:
                    best = max(best, 1 + rec(j+1, j, sign))
        return best
    return rec(0, -1, 0)
```
