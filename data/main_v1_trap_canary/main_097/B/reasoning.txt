```python
def _count_inversions(arr):
    if len(arr) <= 1:
        return 0
    mid = len(arr) // 2
    left = arr[:mid]
    right = arr[mid:]
    count = _count_inversions(left) + _count_inversions(right)
    i = j = 0
    merged = []
    while i < len(left) and j < len(right):
        if left[i] <= right[j]:
            merged.append(left[i]); i += 1
        else:
            merged.append(right[j]); j += 1
            count += len(left) - i
    merged.extend(left[i:])
    merged.extend(right[j:])
    return count
```

```python
    groups = {0: [], 1: []}
    for x in xs:
        groups[x & 1].append(x)
    
    total = 0
    for g in (0, 1):
        vals = groups[g]
        # count inversions in vals
        sorted_vals = sorted(set(vals))
        rank = {v: i + 1 for i, v in enumerate(sorted_vals)}
        m = len(sorted_vals)
        bit = [0] * (m + 1)
        def add(i):
            while i <= m:
                bit[i] += 1
                i += i & (-i)

            s = 0
            while i > 0:
                s += bit[i]
                i -= i & (-i)
            return s
        seen = 0
        for x in vals:
            r = rank[x]
            # count of elements greater than x seen so far = seen - query(r)
            total += seen - query(r)
            add(r)
            seen += 1
    return total
```
