def solve(xs: list[int]) -> int:
    def inversions(a):
        if len(a) <= 1:
            return 0
        mid = len(a)//2
        left = a[:mid]
        right = a[mid:]
        cnt = inversions(left) + inversions(right)
        i = j = 0
        merged = []
        while i < len(left) and j < len(right):
            if left[i] <= right[j]:
                merged.append(left[i]); i += 1
            else:
                cnt += len(left) - i
                merged.append(right[j]); j += 1
        merged.extend(left[i:]); merged.extend(right[j:])
        return cnt
    evens = [x for x in xs if x % 2 == 0]
    odds = [x for x in xs if x % 2 == 1]
    return inversions(evens) + inversions(odds)

def solve(xs: list[int]) -> int:
    count = 0
    n = len(xs)
    for i in range(n):
        for j in range(i + 1, n):

We can process elements from left to right. For each element xs[j], we want to count the number of preceding elements that are greater than xs[j] and have the same parity.

For each parity class, maintain a sorted structure (e.g., a list with bisect) and count the number of preceding elements greater than the current element.

```python
def solve(xs: list[int]) -> int:
    from bisect import bisect_right
    evens = []
    odds = []
    count = 0
    for x in xs:
        lst = evens if x % 2 == 0 else odds
        # count preceding elements in lst greater than x
        # = len(lst) - bisect_right(lst, x)
        count += len(lst) - bisect_right(lst, x)
        # insert x
        insort(lst, x)
    return count
```
