```python
def solve(xs: list[int]) -> int:
    NEG = float('-inf')
    dp = [0, NEG, NEG]
    for x in xs:
        ndp = dp[:]
        for r in range(3):
            if dp[r] != NEG:
                nr = (r + 1) % 3
                v = dp[r] + x
                if v > ndp[nr]:
                    ndp[nr] = v
        dp = ndp
    if dp[2] == NEG:
        return -1
    return dp[2]
```
