```python
def solve(xs: list[int]) -> int:
    if not xs:
        return 0
    max_run = 1
    current_run = 1
    for i in range(1, len(xs)):
        if xs[i] % 2 == xs[i-1] % 2:
            current_run += 1
        else:
            current_run = 1
        max_run = max(max_run, current_run)
    return max_run
```
