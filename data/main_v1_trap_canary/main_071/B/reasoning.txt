```python
def solve(xs: list[int]) -> int:
    vals = sorted(set(xs))
    rank = {v: i+1 for i, v in enumerate(vals)}
    size = len(vals)
    bits = [[0]*(size+1), [0]*(size+1)]
    
    def add(bit, i, delta=1):
        while i <= size:
            bit[i] += delta
            i += i & (-i)
    
    def query(bit, i):
        s = 0
        while i > 0:
            s += bit[i]
            i -= i & (-i)
        return s
    
    total = [0, 0]
    count = 0
    for x in xs:
        r = x & 1  # works for negative numbers too: in Python, -3 & 1 = 1, -2 & 1 = 0. Parity is fine.
        xi = rank[x]
        count += total[r] - query(bits[r], xi)
        add(bits[r], xi)
        total[r] += 1
    return count
```

```python
def solve(xs: list[int]) -> int:
    evens = [x for x in xs if x % 2 == 0]
    odds = [x for x in xs if x % 2 != 0]
    
    def count_inversions(arr):
        # merge sort based
        if len(arr) <= 1:
            return 0, arr
        mid = len(arr) // 2
        left_count, left = count_inversions(arr[:mid])
        right_count, right = count_inversions(arr[mid:])
        count = left_count + right_count
        i = j = 0
        merged = []
        while i < len(left) and j < len(right):
            if left[i] <= right[j]:
                merged.append(left[i])
                i += 1
            else:
                merged.append(right[j])
                j += 1
                count += len(left) - i
        merged.extend(left[i:])
        merged.extend(right[j:])
        return count, merged
    
    return count_inversions(evens)[0] + count_inversions(odds)[0]
```

This is O(n log n). Recursion depth: n can be large, and the recursion depth of merge sort is log2(n), so for reasonable n it's fine. However, if n is huge (like 10^6), log2(10^6) ≈ 20, so no problem.

Note on negative numbers: in Python, x % 2 == 0 works correctly for negative numbers too (e.g., -3 % 2 = 1, -4 % 2 = 0). Good.
